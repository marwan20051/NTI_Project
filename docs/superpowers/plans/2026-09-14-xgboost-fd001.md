```python
from pathlib import Path

import pandas as pd
from sklearn.model_selection import GroupShuffleSplit

CMAPSS_COLUMNS = (
    ["unit_id", "cycle"]
    + [f"setting_{index}" for index in range(1, 4)]
    + [f"sensor_{index}" for index in range(1, 22)]
)


def load_sensor_file(path: Path) -> pd.DataFrame:
    if not path.is_file():
        raise FileNotFoundError(f"Missing C-MAPSS file: {path}")
    frame = pd.read_csv(path, sep=r"\s+", header=None, names=CMAPSS_COLUMNS)
    if frame.shape[1] != len(CMAPSS_COLUMNS) or frame.isna().all(axis=0).any():
        raise ValueError(f"Invalid C-MAPSS schema in {path}")
    frame[["unit_id", "cycle"]] = frame[["unit_id", "cycle"]].astype(int)
    return frame


def add_train_rul(frame: pd.DataFrame) -> pd.DataFrame:
    result = frame.copy()
    maximum_cycle = result.groupby("unit_id")["cycle"].transform("max")
    result["rul"] = maximum_cycle - result["cycle"]
    if result["rul"].lt(0).any():
        raise ValueError("Training RUL cannot be negative")
    return result


def last_cycle_rows(frame: pd.DataFrame) -> pd.DataFrame:
    indices = frame.groupby("unit_id")["cycle"].idxmax()
    return frame.loc[indices].sort_values("unit_id").reset_index(drop=True)


def attach_test_rul(frame: pd.DataFrame, labels: pd.Series) -> pd.DataFrame:
    result = last_cycle_rows(frame)
    clean_labels = pd.Series(labels).reset_index(drop=True).astype(float)
    if len(result) != len(clean_labels):
        raise ValueError("Expected one RUL value per test engine")
    result["rul"] = clean_labels.to_numpy()
    return result


def split_by_engine(frame, validation_size=0.2, random_state=42):
    splitter = GroupShuffleSplit(
        n_splits=1, test_size=validation_size, random_state=random_state
    )
    train_index, valid_index = next(splitter.split(frame, groups=frame["unit_id"]))
    train = frame.iloc[train_index].copy()
    valid = frame.iloc[valid_index].copy()
    if not set(train.unit_id).isdisjoint(set(valid.unit_id)):
        raise RuntimeError("Engine leakage detected")
    return train, valid


def load_fd001(data_dir: Path):
    train = add_train_rul(load_sensor_file(data_dir / "train_FD001.txt"))
    test = load_sensor_file(data_dir / "test_FD001.txt")
    rul_path = data_dir / "RUL_FD001.txt"
    if not rul_path.is_file():
        raise FileNotFoundError(f"Missing C-MAPSS file: {rul_path}")
    labels = pd.read_csv(rul_path, sep=r"\s+", header=None).iloc[:, 0]
    test_last = attach_test_rul(test, labels)
    return train, test, test_last
```

```python
import numpy as np
import pandas as pd


def select_sensors(train_frame, top_n=8, variance_floor=1e-10):
    sensors = [column for column in train_frame if column.startswith("sensor_")]
    variable = [column for column in sensors if train_frame[column].var() > variance_floor]
    correlations = train_frame[variable].corrwith(train_frame["rul"]).abs()
    return correlations.sort_values(ascending=False).head(top_n).index.tolist()


def _rolling_slope(values):
    values = np.asarray(values, dtype=float)
    if len(values) < 2:
        return 0.0
    x = np.arange(len(values), dtype=float)
    denominator = ((x - x.mean()) ** 2).sum()
    return 0.0 if denominator == 0 else float(((x - x.mean()) * (values - values.mean())).sum() / denominator)


def engineer_features(frame, selected_sensors, windows=(5, 15)):
    ordered = frame.sort_values(["unit_id", "cycle"]).copy()
    base_columns = ["cycle", "setting_1", "setting_2", "setting_3", *selected_sensors]
    output = ordered[base_columns].copy()
    grouped = ordered.groupby("unit_id", sort=False)
    for sensor in selected_sensors:
        output[f"{sensor}_diff_1"] = grouped[sensor].diff().fillna(0.0)
        for window in windows:
            rolling = grouped[sensor].rolling(window, min_periods=1)
            output[f"{sensor}_roll_mean_{window}"] = rolling.mean().reset_index(level=0, drop=True)
            output[f"{sensor}_roll_std_{window}"] = rolling.std(ddof=0).reset_index(level=0, drop=True).fillna(0.0)
            output[f"{sensor}_slope_{window}"] = (
                rolling.apply(_rolling_slope, raw=True).reset_index(level=0, drop=True).fillna(0.0)
            )
    output = output.replace([np.inf, -np.inf], 0.0).fillna(0.0)
    return output.loc[frame.index]
```

import numpy as np
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score


def nasa_score(y_true, y_pred):
    error = np.asarray(y_pred, dtype=float) - np.asarray(y_true, dtype=float)
    penalties = np.where(error < 0, np.exp(-error / 13.0) - 1.0, np.exp(error / 10.0) - 1.0)
    return float(penalties.sum())


def regression_metrics(y_true, y_pred):
    y_true = np.asarray(y_true, dtype=float)
    y_pred = np.clip(np.asarray(y_pred, dtype=float), 0.0, None)
    return {
        "rmse": float(mean_squared_error(y_true, y_pred) ** 0.5),
        "mae": float(mean_absolute_error(y_true, y_pred)),
        "r2": float(r2_score(y_true, y_pred)),
        "nasa_score": nasa_score(y_true, y_pred),
    }
```

import numpy as np
import xgboost as xgb


def resolve_device(prefer_gpu=True):
    if not prefer_gpu:
        return "cpu"
    build_info = xgb.build_info() if hasattr(xgb, "build_info") else {}
    if not build_info.get("USE_CUDA", False):
        return "cpu"
    try:
        probe = xgb.XGBRegressor(n_estimators=1, max_depth=1, tree_method="hist", device="cuda")
        probe.fit(np.array([[0.0], [1.0]]), np.array([0.0, 1.0]), verbose=False)
        return "cuda"
    except (xgb.core.XGBoostError, ValueError):
        return "cpu"


def _common(device):
    return {
        "objective": "reg:squarederror",
        "tree_method": "hist",
        "device": device,
        "random_state": 42,
        "n_jobs": -1,
    }


def baseline_parameters(device):
    return {
        **_common(device),
        "n_estimators": 350,
        "max_depth": 6,
        "learning_rate": 0.08,
        "subsample": 0.9,
        "colsample_bytree": 0.9,
        "reg_lambda": 1.0,
    }


def improved_parameters(device):
    return {
        **_common(device),
        "n_estimators": 1200,
        "max_depth": 4,
        "learning_rate": 0.025,
        "min_child_weight": 5,
        "subsample": 0.85,
        "colsample_bytree": 0.8,
        "reg_alpha": 0.05,
        "reg_lambda": 5.0,
        "early_stopping_rounds": 75,
    }
```
